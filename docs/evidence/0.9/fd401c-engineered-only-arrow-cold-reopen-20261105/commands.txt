export JAVA_HOME=/usr/lib/jvm/java-21-openjdk
export PATH="$JAVA_HOME/bin:$PATH"
./gradlew createClientLaunchScript \
  -PfdBuildDirectory=/tmp/fd401c-engineered-only-cold-reopen-20261105/build \
  -PfdPackaged -PfdIntegrations=engineered -PfdFeNetwork -PfdValidation \
  -PfdWorld=FD401Stop12 -PfdClientMaxHeap=1536M \
  -PfdClientGameDirectory=/tmp/fd401c-engineered-only-cold-reopen-20261105/game \
  -x jar -x compileJava --no-daemon --offline --console=plain \
  -Dorg.gradle.jvmargs=-Xmx1G

# The generated direct client script did not encode an Xmx itself, so the task-owned
# wrapper adds exactly -Xmx1536M to its Java 21 invocation. See run-client-1536M.sh.

# Open 1: private X server stayed alive through complete client exit.
systemd-run --user --unit=fd401c-engineered-cold-xvfb352.service \
  --property=Restart=no /usr/bin/Xvfb :352 -screen 0 1280x720x24 -nolisten tcp
systemd-run --user --scope --unit=fd401c-engineered-cold-client.scope \
  -p MemoryMax=3221225472 -- env -u WAYLAND_DISPLAY DISPLAY=:352 \
  XDG_SESSION_TYPE=x11 ALSOFT_DRIVERS=null JAVA_HOME=/usr/lib/jvm/java-21-openjdk \
  PATH=/usr/lib/jvm/java-21-openjdk/bin:$PATH \
  /tmp/fd401c-engineered-only-cold-reopen-20261105/run-client-1536M.sh
