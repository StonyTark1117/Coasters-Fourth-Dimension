from pathlib import Path
import subprocess,time,sys,datetime
mode=sys.argv[1]
out=Path('/tmp/fd-transport-'+mode+'-commands.txt')
with out.open('w') as f:
 def cmd(*args):
  f.write(datetime.datetime.now().isoformat()+'\n');f.flush()
  subprocess.run(['python3','/tmp/fd-combined-rcon.py',*args],stdout=f,stderr=subprocess.STDOUT,check=True)
 cmd('list','tp Dev -5.5 180 17.5 -90 -12')
 time.sleep(8)
 for i in range(6):
  cmd('tp Dev 5000 180 5000 0 0');time.sleep(8)
  cmd('tp Dev -5.5 180 17.5 -90 -12');time.sleep(8)
 cmd('tick query','save-all flush')
Path('/tmp/fd-transport-'+mode+'-transitions.exit').write_text('0\n')
