/** Native ordinary-bearing access, separate from 4D factory access and recovery towing. */
@GameTestHolder("fourthdimension")
@PrefixGameTestTemplate(false)
public final class OrdinaryAccessGameTests {
  private static final List<TrackAppearance> APPEARANCES =
      List.of(
          TrackAppearance.ORIGINAL,
          new TrackAppearance(
              AppearanceProviders.TRACK_STYLES, "coasterssimulatedextratypes:arrow_4d"),
          new TrackAppearance(
              AppearanceProviders.TRACK_STYLES, "coasterssimulatedextratypes:arrow_4d_dbl_spine"),
          new TrackAppearance("missing:appearance", "missing:style"));

  @GameTest(template = "empty", timeoutTicks = 160)
  public static void fiveFamilyOrdinaryFitSnapAndTraversal(GameTestHelper helper) {
    var level = helper.getLevel();
    var data = FoundationData.get(level);
    var container = SubLevelContainer.getContainer(level);
    var physics = container.physicsSystem().getPipeline();
    var player =
        FakePlayerFactory.get(
            level,
            new com.mojang.authlib.GameProfile(
                UUID.fromString("fdfdfdfd-0000-0000-0000-000000001108"), "FDOrdinaryMatrix"));
    player.setGameMode(GameType.CREATIVE);
    var rails = new ArrayList<FoundationData.Rail>();
    var bodies = new ArrayList<ServerSubLevel>();
    var hits = new ArrayList<CoasterPathTrackFrame.GraphHit>();
    var starts = new ArrayList<Vector3d>();
    var turns = new ArrayList<Vector3d>();
    for (int x = -51; x <= -43; x++)
      for (int z = 39; z <= 44; z++) level.setChunkForced(x, z, true);
    for (var design : TrackDesign.values()) {
      var from = new BlockPos(-800 + design.ordinal() * 24, 120, 640);
      var to = from.offset(0, 0, 48);
      for (var anchor : List.of(from, to))
        level.setBlockAndUpdate(
            anchor, SimulatedCoastersBlocks.COASTER_ANCHORPOINT.get().defaultBlockState());
      player.setPos(to.getX() + 2, to.getY() + 1, to.getZ());
      player.setItemInHand(InteractionHand.MAIN_HAND, TrackItems.ARROW.get().getDefaultInstance());
      rails.add(TrackPlacement.connect(level, from, to, design, player, player.getMainHandItem()));
    }
    helper.runAtTickTime(
        20,
        () -> {
          for (var rail : rails) {
            var edge = TrackGeometry.resolve(level, rail);
            var hit =
                new CoasterPathTrackFrame.GraphHit(
                    CoasterPathGraphManager.sampleEdgePosition(level, edge, .5), edge, .5);
            hits.add(hit);
            var orientation =
                CoasterPathTrackFrame.quaternionFromTrackBasis(
                    CoasterPathTrackFrame.trackBasisAtHit(level, hit));
            var origin =
                CoasterCartLinkPlacement.plotOriginForBearingCenter(
                    CoasterPathTrackFrame.pathAnchorForSnappedBlock(level, hit), orientation);
            bodies.add(
                CoasterCartSpawner.spawnMinimalContraption(level, origin, orientation, hit, true));
          }
        });
    // The native representative-bearing index is populated by the next physics update.
    helper.runAtTickTime(
        25,
        () -> {
          for (int i = 0; i < rails.size(); i++) {
            var rail = rails.get(i);
            var body = bodies.get(i);
            var hit = hits.get(i);
            var bearing = CoasterCartTrackSnap.representativeBearingPlotPos(body);
            helper.assertTrue(bearing != null, "Ordinary fixture has no native bearing");
            int index = data.rails.indexOf(rail);
            for (int appearance = 0; appearance < APPEARANCES.size(); appearance++) {
              rail.appearance = APPEARANCES.get(appearance);
              helper.assertTrue(
                  OrdinaryGearCertificate.fits(body), "Bare native bearing lost certification");
              decision(helper, body, hit, rail, appearance, "bare", true);
              for (var extra : List.of(bearing.above(), bearing.east())) {
                level.setBlockAndUpdate(extra, Blocks.STONE.defaultBlockState());
                try {
                  helper.assertTrue(
                      !OrdinaryGearCertificate.fits(body),
                      "Added cart body passed the gear certificate");
                  decision(
                      helper,
                      body,
                      hit,
                      rail,
                      appearance,
                      extra.equals(bearing.above()) ? "body_above" : "body_beside",
                      false);
                  if (extra.equals(bearing.above())) {
                    data.rails.remove(index);
                    try {
                      decision(helper, body, hit, rail, appearance, "ordinary_track", true);
                    } finally {
                      data.rails.add(index, rail);
                    }
                  }
                } finally {
                  level.removeBlock(extra, false);
                }
              }
              rail.valid = false;
              try {
                decision(helper, body, hit, rail, appearance, "invalid_rail", false);
              } finally {
                rail.valid = true;
              }
              for (var provider : List.of("missing:functional", AppearanceProviders.TRACK_STYLES)) {
                var unavailable =
                    new FoundationData.Rail(
                        rail.from,
                        rail.to,
                        rail.family,
                        rail.profile,
                        rail.id,
                        rail.variant,
                        rail.geometryHash,
                        rail.revision,
                        true,
                        provider);
                unavailable.appearance = rail.appearance;
                data.rails.set(index, unavailable);
                try {
                  decision(helper, body, hit, rail, appearance, provider, false);
                } finally {
                  data.rails.set(index, rail);
                }
              }
            }
            rail.appearance = TrackAppearance.ORIGINAL;
            CoasterCartTrackSnap.applyInitialSnapAfterSpawn(body, level, hit);
          }
        });
    helper.runAtTickTime(
        30,
        () -> {
          for (var body : bodies) {
            starts.add(new Vector3d(body.logicalPose().position()));
            physics.addLinearAndAngularVelocity(
                body,
                new Vector3d(0, 0, 3).sub(physics.getLinearVelocity(body, new Vector3d())),
                new Vector3d());
          }
        });
    helper.runAtTickTime(
        55,
        () -> {
          for (int i = 0; i < bodies.size(); i++) {
            var body = bodies.get(i);
            var position = new Vector3d(body.logicalPose().position());
            traversal(helper, body, rails.get(i), starts.get(i), position, true);
            turns.add(position);
            physics.addLinearAndAngularVelocity(
                body,
                new Vector3d(0, 0, -3).sub(physics.getLinearVelocity(body, new Vector3d())),
                new Vector3d());
          }
        });
    helper.runAtTickTime(
        80,
        () -> {
          for (int i = 0; i < bodies.size(); i++) {
            var body = bodies.get(i);
            traversal(
                helper,
                body,
                rails.get(i),
                turns.get(i),
                new Vector3d(body.logicalPose().position()),
                false);
            container.removeSubLevel(body, SubLevelRemovalReason.REMOVED);
          }
          for (var rail : rails) {
            ((CoasterAnchorpointBlockEntity) level.getBlockEntity(rail.from))
                .removeAnchorPeerCurve(level, rail.to);
            data.rails.remove(rail);
            level.removeBlock(rail.from, false);
            level.removeBlock(rail.to, false);
          }
          System.out.println(
              "FD_ORDINARY_MATRIX families=5 appearances=4 accepted=40 rejected=100 traversals=10"
                  + " registered4D=0");
          helper.succeed();
        });
  }

  private static void decision(
      GameTestHelper helper,
      ServerSubLevel body,
      CoasterPathTrackFrame.GraphHit hit,
      FoundationData.Rail rail,
      int appearance,
      String condition,
      boolean expected) {
    var level = helper.getLevel();
    String label = rail.family.id() + "/" + appearance + "/" + condition;
    helper.assertTrue(
        EligibilityHooks.allows(body, level, hit) == expected, "Eligibility mismatch " + label);
    helper.assertTrue(
        (CoasterCartRailSnapEligibility.filterHitForBearing(body, level, hit) != null) == expected,
        "Native filter mismatch " + label);
    var center =
        net.minecraft.world.phys.Vec3.atCenterOf(
            CoasterCartTrackSnap.representativeBearingPlotPos(body));
    var before = body.logicalPose().transformPosition(center);
    EligibilityHooks.bindAccepted(body, level, hit);
    CoasterCartTrackSnap.applyInitialSnapAfterSpawn(body, level, hit);
    if (expected) {
      var bearing = CoasterCartTrackSnap.representativeBearingPlotPos(body);
      helper.assertTrue(
          level.getBlockState(bearing).getValue(CoasterCartBlock.SNAP_ENGAGED),
          "Native initial snap rejected " + label);
    } else {
      helper.assertTrue(
          CoasterCartTrackSnap.isBearingDisengaged(body)
              && before.distanceTo(body.logicalPose().transformPosition(center)) < 1e-8,
          "Rejected initial snap changed engagement or bearing position "
              + label
              + " disengaged="
              + CoasterCartTrackSnap.isBearingDisengaged(body)
              + " before="
              + before
              + " after="
              + body.logicalPose().transformPosition(center));
    }
    helper.assertTrue(
        !FoundationData.get(level).vehicles.containsKey(body.getUniqueId())
            && FamilyControllers.find(body).isEmpty(),
        "Ordinary body acquired 4D behavior " + label);
    System.out.println(
        "FD_ORDINARY_DECISION family="
            + rail.family.id()
            + " appearance="
            + appearance
            + " condition="
            + condition
            + " accepted="
            + expected
            + " nativeFilter=true initialSnap=true bindingUnregistered=true");
  }

  private static void traversal(
      GameTestHelper helper,
      ServerSubLevel body,
      FoundationData.Rail rail,
      Vector3d before,
      Vector3d after,
      boolean forward) {
    double progress = (after.z - before.z) * (forward ? 1 : -1);
    double lateral = Math.abs(after.x - before.x);
    double vertical = Math.abs(after.y - before.y);
    helper.assertTrue(
        progress > .5 && lateral < .2 && vertical < .2,
        "Ordinary bearing failed guided traversal "
            + rail.family.id()
            + "/"
            + forward
            + ": "
            + before
            + " -> "
            + after);
    helper.assertTrue(
        !CoasterCartTrackSnap.isBearingDisengaged(body)
            && !FoundationData.get(helper.getLevel()).vehicles.containsKey(body.getUniqueId()),
        "Moving ordinary bearing disengaged or acquired 4D behavior");
    System.out.println(
        "FD_ORDINARY_TRAVERSAL family="
            + rail.family.id()
            + " direction="
            + (forward ? "forward" : "reverse")
            + " progress="
            + progress
            + " lateral="
            + lateral
            + " vertical="
            + vertical
            + " engaged=true registered4D=false");
  }
}
