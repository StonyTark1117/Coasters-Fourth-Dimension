def warnings(text):
 result=set()
 for line in text.splitlines():
  if '/WARN]' not in line and '/ERROR]' not in line:continue
  line=re.sub(r'^\[[^]]+\] ', '', line)
  line=re.sub(r'[0-9a-f]{8}-[0-9a-f-]{27,}', '<UUID>', line)
  result.add(line)
 return result
