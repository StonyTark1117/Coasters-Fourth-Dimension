from pathlib import Path
import subprocess,time,os,json,hashlib,shutil
root=Path(__file__).resolve().parents[2];out=Path(__file__).resolve().parent
source=root/'src/main/java/dev/fourthdimension/runtime/EligibilityHooks.java'
original=source.read_bytes()
assert original==subprocess.check_output(['git','show','HEAD:src/main/java/dev/fourthdimension/runtime/EligibilityHooks.java'],cwd=root)
def digest(b):return hashlib.sha256(b).hexdigest()
def run(name,profile=None,build=False):
 cmd=['./gradlew','-Dorg.gradle.jvmargs=-Xmx1G','--max-workers=2']
 if build:cmd.append('build')
 cmd.append('runGameTestServer')
 if profile:cmd.append('-PfdIntegrations='+profile)
 if name in ('engineered','combined'):cmd.append('-PfdPaxEnergyReload')
 with (out/(name+'.log')).open('w') as log:
  p=subprocess.Popen(cmd,cwd=root,stdin=subprocess.DEVNULL,stdout=log,stderr=subprocess.STDOUT,env={**os.environ,'JAVA_HOME':'/usr/lib/jvm/java-21-openjdk','JAVA_TOOL_OPTIONS':'-Xmx2G'})
  (out/'matrix-status.json').write_text(json.dumps({'status':'running','profile':name,'pid':p.pid,'command':cmd})+'\n');code=p.wait()
 (out/(name+'.exit')).write_text(str(code)+'\n');return code
try:
 while not (out/'base.exit').exists():
  os.kill(int((out/'base.pid').read_text()),0);time.sleep(2)
 assert (out/'base.exit').read_text().strip()=='0','Base fixture failed'
 (out/'production-original.java.txt').write_bytes(original)
 old='''        || rail.family() != vehicle.family
        || !rail.variant.equals(vehicle.variant)) return false;'''
 replacement='''        || (vehicle.family == dev.fourthdimension.contract.Family.ARROW
            && (rail.family() != vehicle.family || !rail.variant.equals(vehicle.variant)))) return false;'''
 assert original.decode().count(old)==1
 mutant=original.decode().replace(old,replacement).encode();(out/'mutation.java.txt').write_bytes(mutant)
 (out/'mutation.json').write_text(json.dumps({'scope':'Negative control only: bypass family/variant checks for non-Arrow cars; retain route adjacency, provider and consist checks','original_sha256':digest(original),'mutant_sha256':digest(mutant)},indent=2)+'\n')
 try:
  source.write_bytes(mutant)
  code=run('mutant');assert code==1,'Guard mutation failed to trigger the regression'
  assert 'Connected family eligibility mismatch ZACSPIN/ARROW' in (out/'mutant.log').read_text()
  compiled=Path('/tmp/fourthdimension-48535694/build/classes/java/main/dev/fourthdimension/runtime/EligibilityHooks.class')
  (out/'mutant-class.sha256').write_text(digest(compiled.read_bytes())+'\n')
 finally:
  source.write_bytes(original)
  (out/'restored.json').write_text(json.dumps({'original_sha256':digest(original),'restored_sha256':digest(source.read_bytes())},indent=2)+'\n')
 for name,profile in [('styles','styles'),('engineered','engineered'),('combined','engineered,styles,additions,magnetized,immersion'),('additions','additions'),('magnetized','magnetized'),('immersion','immersion')]:
  assert run(name,profile,build=name=='styles')==0,'Native profile failed: '+name
  if name=='styles':shutil.copyfile('/tmp/fourthdimension-48535694/build/libs/coasters-fourth-dimension-0.9.0-dev.1.jar',out/'fixed.jar')
 assert source.read_bytes()==original
 (out/'matrix-status.json').write_text(json.dumps({'status':'complete'})+'\n');(out/'matrix.exit').write_text('0\n')
except BaseException as e:
 (out/'matrix-error.txt').write_text(repr(e)+'\n');(out/'matrix.exit').write_text('1\n');raise
