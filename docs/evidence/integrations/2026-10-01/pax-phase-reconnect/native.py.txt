from pathlib import Path
import subprocess,os,json
root=Path(__file__).resolve().parents[2];out=Path(__file__).resolve().parent
try:
 for name,profile in [('native-base',None),('native-engineered','engineered'),('native-combined','engineered,styles,additions,magnetized,immersion')]:
  cmd=['./gradlew','-Dorg.gradle.jvmargs=-Xmx1G','--max-workers=2','runGameTestServer']
  if profile:cmd+=['-PfdIntegrations='+profile,'-PfdPaxEnergyReload']
  with (out/(name+'.log')).open('w') as log:
   p=subprocess.Popen(cmd,cwd=root,stdin=subprocess.DEVNULL,stdout=log,stderr=subprocess.STDOUT,env={**os.environ,'JAVA_HOME':'/usr/lib/jvm/java-21-openjdk','JAVA_TOOL_OPTIONS':'-Xmx2G'})
   (out/'native-status.json').write_text(json.dumps({'profile':name,'pid':p.pid,'status':'running'})+'\n');code=p.wait()
  (out/(name+'.exit')).write_text(str(code)+'\n');assert code==0,name
 (out/'native-status.json').write_text(json.dumps({'status':'complete'})+'\n');(out/'native.exit').write_text('0\n')
except BaseException as e:
 (out/'native-error.txt').write_text(repr(e)+'\n');(out/'native.exit').write_text('1\n');raise
